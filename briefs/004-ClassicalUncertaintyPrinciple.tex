\begin{abstract}
	We show that a zero lower bound on physical entropy, motivated by the third law of thermodynamics, gives rise to the classical uncertainty relation $\sigma_q\sigma_p\geq\frac{\hbar}{e}$. For non-conjugate variables, the relation becomes $\sigma_f\sigma_g\geq\frac{\hbar}{e}e^{E[\ln|\{f,g\}|]}$, where $E[\ln|\{f,g\}|]$ is the expectation of the logarithm of the absolute value of the Poisson bracket.
\end{abstract}

\section{Introduction}

The third law of thermodynamics provides a physical reference for zero entropy. Here we show that imposing a non-negative entropy in classical statistical mechanics gives a classical version of the quantum uncertainty relation:
\begin{equation}
	\sigma_q\sigma_p\geq\frac{h}{2\pi e}=\frac{\hbar}{e}.
\end{equation}
In this context, $h$ is the phase-space area for which a uniform distribution has zero entropy, and $\hbar=\frac{h}{2\pi}$. The $2\pi e$ factor appears because the product of two independent Gaussian distributions minimizes the uncertainty at fixed entropy.

For convenience, entropy is measured in nats, meaning that we use the natural logarithm and set the Boltzmann constant to $k_B=1$. Since we are looking at zero entropy, a change of entropy units does not change the result.

We also establish the converse: if the generalized uncertainty relation holds in every admissible coordinate system, the entropy must be non-negative. Both directions follow from an exact equality separating the entropy, the KL divergence to a product Gaussian, and the coordinate adjustment. Suitable coordinates make the KL contribution vanish or approach zero, allowing the uncertainty bound to recover the entropy bound. The physical results are stated under a separate requirement that ensembles have finite entropy. The mathematical extension to measurable densities also treats well-defined infinite entropy; the additional issues are discussed separately.

\section{Mathematical prerequisites}

First some notation. Since we need to transform the same distribution over different coordinates, the functions $\rho_{qp}(q,p)$ and $\rho_{fg}(f,g)$, shortened to $\rho_{qp}$ and $\rho_{fg}$, represent the same distribution as expressed in different coordinates. The Shannon differential entropy $S[\rho_{qp}]=-\int\rho_{qp}\ln(h \rho_{qp})\,dqdp$ is calculated on the given coordinates. Similarly for expectations $E[A\mid\rho_{qp}]=\int A(q,p)\rho_{qp}\,dqdp$. Where indicated, infinite variance or entropy is allowed. We use the extended variance $\sigma_X^2=\inf_{a\in\mathbb{R}}E[(X-a)^2\mid\rho_{qp}]\in[0,+\infty]$, which remains defined even when the mean does not exist.

\begin{defn}
	For a distribution $\rho$ with density $\rho_{qp}$, an admissible coordinate transformation is a $C^1$ diffeomorphism $T=(f,g):\mathbb{R}^2\to U$ onto an open region $U\subseteq\mathbb{R}^2$, with $C^1$ inverse and $E[|\ln J_T|\mid\rho_{qp}]<\infty$, where $J_T=|\det DT|$ is the absolute Jacobian determinant.
\end{defn}
The region $U$ may be bounded. A coordinate density on $U$ is extended by zero when regarded as a function on $\mathbb{R}^2$.

This first proposition quantifies the uncertainty relationship in a generic set of coordinates $(f,g)$ based on the entropy in a reference coordinate system $(q,p)$, the entropy change due to the coordinate change and the entropy difference from a Gaussian with matching variances, since the Gaussian minimizes the spread at fixed entropy.

\begin{prop}\label{entropyVarianceIdentity}
	Let $\rho_{qp}(q,p)$ be a probability density with finite entropy $S[\rho_{qp}]$. Let $T: (q,p) \mapsto (f(q,p),g(q,p))$ be an admissible coordinate change such that $\rho_{fg}(f,g)$ has finite variances $(\sigma_f^2,\sigma_g^2)$. Let $G_{fg}$ be the product of two Gaussians with the same means and variances. Then
	\begin{equation}\label{entropyVarianceEquality}
		\sigma_f\sigma_g=\frac{h}{2\pi e}e^{S[\rho_{qp}]}e^{E[\ln J_T\mid\rho_{qp}]}e^{D(\rho_{fg}\Vert G_{fg})}
	\end{equation}
	where $J_T$ is the absolute Jacobian determinant of $T$.
\end{prop}

\begin{proof}
	Admissibility and the standard coordinate-transformation law for differential entropy \cite[Lemma 2]{RioulCosta} ensure that $S[\rho_{fg}]$ is finite as well. Adding and subtracting the intermediate entropies gives
	\begin{equation*}
		S[G_{fg}]=S[\rho_{qp}]+\bigl(S[\rho_{fg}]-S[\rho_{qp}]\bigr)+\bigl(S[G_{fg}]-S[\rho_{fg}]\bigr).
	\end{equation*}
	The first difference is the coordinate contribution $E[\ln J_T\mid\rho_{qp}]$. For the second difference, note that, since $G_{fg}$ is a Gaussian with the same mean and variance of $\rho_{fg}$, calculations show that $\int \rho_{fg} \ln G_{fg} df dg = \int G_{fg}  \ln G_{fg} df dg$. Therefore $S[G_{fg}]-S[\rho_{fg}] = D(\rho_{fg}\Vert G_{fg})$ is the Kullback--Leibler (KL) divergence between the two distributions.

	Substituting these contributions and the Gaussian entropy formula \cite[Section 8.4]{CoverThomas} gives
	\begin{equation*}
		\ln\left(\frac{2\pi e\,\sigma_f\sigma_g}{h}\right)=S[G_{fg}]=S[\rho_{qp}]+E[\ln J_T\mid\rho_{qp}]+D(\rho_{fg}\Vert G_{fg}).
	\end{equation*}
	Exponentiating proves \eqref{entropyVarianceEquality}.
\end{proof}

The next proposition is based on the result that any smooth probability density that is nowhere zero can be transformed into a zero entropy Gaussian with a smooth transformation. The entropy in the reference coordinate system, then, becomes equal to minus the entropy change induced by the transformation.

\begin{prop}\label{smoothEntropyCoordinates}
	Let $\rho_{qp}(q,p)$ be a positive smooth probability density with finite $S[\rho_{qp}]$. There is a smooth admissible coordinate change for which $\rho_{fg}$ is a product Gaussian with $S[\rho_{fg}]=0$. In these coordinates, $\sigma_f\sigma_g=\frac{h}{2\pi e}$ and $S[\rho_{qp}]=-E[\ln J_T\mid\rho_{qp}]$.
\end{prop}

\begin{proof}
	Choose $s,t>0$ with $st=\frac{h}{2\pi e}$, and let $G_{fg}(f,g)=\frac{1}{2\pi st}e^{-\frac{f^2}{2s^2}}e^{-\frac{g^2}{2t^2}}$. Its entropy is $S[G_{fg}]=\ln\left(\frac{2\pi e\,st}{h}\right)=0$. By the smooth volume-transport theorem of Greene and Shiohama \cite[Theorem 1]{GreeneShiohama}, there is a smooth diffeomorphism $T:\mathbb{R}^2\to\mathbb{R}^2$ with $\rho_{qp}(q,p)=G_{fg}(T(q,p))J_T(q,p)$. The two volume forms have the same total mass, and every end has finite volume because that total mass is finite. Thus $\rho_{fg}=G_{fg}$.

	Finite $S[\rho_{qp}]$ gives $E[|\ln\rho_{qp}|\mid\rho_{qp}]<\infty$. Also $E[|\ln G_{fg}(T)|\mid\rho_{qp}]=E[|\ln G_{fg}|\mid G_{fg}]<\infty$, so
	\begin{equation*}
		E[|\ln J_T|\mid\rho_{qp}]\leq E[|\ln\rho_{qp}|\mid\rho_{qp}]+E[|\ln G_{fg}|\mid G_{fg}]<\infty.
	\end{equation*}
	Thus $T$ is admissible. The coordinate entropy formula gives $0=S[\rho_{fg}]=S[\rho_{qp}]+E[\ln J_T\mid\rho_{qp}]$, proving the result.
\end{proof}

The next proposition extends the result to measurable probability densities, including well-defined infinite entropy. In the finite-entropy case, the KL contribution can be made arbitrarily small. For an admissible coordinate change, write $U_\rho(T)=\ln\left(\frac{2\pi e\,\sigma_f\sigma_g}{h}\right)-E[\ln J_T\mid\rho_{qp}]$, with $U_\rho(T)=+\infty$ if a variance is infinite. When the variances and $S[\rho_{qp}]$ are finite, Proposition \ref{entropyVarianceIdentity} gives
\begin{equation*}
	U_\rho(T)=S[\rho_{qp}]+D(\rho_{fg}\Vert G_{fg})\geq S[\rho_{qp}].
\end{equation*}

\begin{prop}\label{measurableEntropyCoordinates}
	Let $\rho_{qp}(q,p)$ be a probability density with well-defined entropy $S[\rho_{qp}]$, including extended values. Then
	\begin{equation}\label{entropyCoordinateInfimum}
		\inf_T U_\rho(T)=S[\rho_{qp}],
	\end{equation}
	where the infimum is over smooth admissible coordinate changes. If $S[\rho_{qp}]$ is finite, there are smooth admissible $T_j=(f_j,g_j):\mathbb{R}^2\to\mathbb{R}^2$ with finite coordinate variances and $D(\rho_{f_jg_j}\Vert G_j)\to0$, where $G_j$ is the product Gaussian matching the separate means and variances of $\rho_{f_jg_j}$. In particular, $U_\rho(T_j)\to S[\rho_{qp}]$.
\end{prop}

\noindent\emph{Proof sketch.}
For finite $S[\rho_{qp}]$, choose positive smooth coordinate densities $\eta_j$ on the $q,p$ plane such that $d_j=D(\rho_{qp}\Vert\eta_j)\to0$ and $E[|\ln\eta_j|\mid\rho_{qp}]<\infty$.

Transport each $\eta_j$ to the same product Gaussian $\gamma_{fg}$ on $\mathbb{R}^2$. Invariance of relative entropy gives $D(\rho_{f_jg_j}\Vert\gamma_{fg})=d_j$. The entropy variational inequality ensures finite coordinate variances; together with the logarithmic control of $\eta_j$, this also ensures admissibility. Let $G_j$ be the product Gaussian matching the separate means and variances of $\rho_{f_jg_j}$. The Gaussian moment identity gives $0\leq D(\rho_{f_jg_j}\Vert G_j)\leq d_j\to0$. Proposition \ref{entropyVarianceIdentity} then gives
\begin{equation*}
	U_\rho(T_j)=S[\rho_{qp}]+D(\rho_{f_jg_j}\Vert G_j)\longrightarrow S[\rho_{qp}].
\end{equation*}
Together with the lower bound, this proves \eqref{entropyCoordinateInfimum} for finite entropy.

If $S[\rho_{qp}]=-\infty$, the same approximation provides smooth densities $\eta_j$ with finite logarithmic expectations tending to $+\infty$. Transporting them to a fixed uniform density on a bounded rectangle keeps the variances bounded while $E[\ln J_{T_j}\mid\rho_{qp}]\to+\infty$, so $U_\rho(T_j)\to-\infty$. If $S[\rho_{qp}]=+\infty$, every admissible frame retains infinite differential entropy and must have an infinite variance. Thus $U_\rho(T)=+\infty$ in every frame.

\section{Conditions}

We now state the conditions for the theorems, starting with the classical phase-space representation. This includes pure states, ensembles, configuration counting and entropy.

\begin{condition}[Classical states]{CST}
	The state space of a classical system with $n$ degrees of freedom (DOFs) is represented by $M=\mathbb{R}^{2n}$ with canonical coordinates $(q^i,p_i)$ and symplectic form $\omega=\sum_i dq^i\wedge dp_i$. The count of configurations for each DOF is represented by its canonical phase-space area, and the count of states by the Liouville measure $\mu(U)=\int_U dq^1\cdots dq^n dp_1\cdots dp_n=\int_U\frac{\omega^n}{n!}$.

	The space of statistical ensembles is given by the convex space of probability densities $\rho(q^i,p_i)$. The Gibbs/Shannon entropy is $S[\rho]=-\int_M\rho_{qp}\ln(h^n\rho_{qp})\,d\mu=S[\rho_{qp}]$, where $h>0$ is a fixed constant with units of phase-space area.
\end{condition}

The constant $h$ makes the entropy formula dimensionally correct and fixes its zero. For an ensemble $\rho_U$ uniform over a region $U$ with $0<\mu(U)<\infty$, the entropy is $S[\rho_U]=\ln\left(\frac{\mu(U)}{h^n}\right)$, which vanishes exactly when its phase-space volume is $h^n$.

The representation alone does not guarantee that this entropy is finite or even well-defined. We therefore impose finiteness as a separate physical requirement.

\begin{condition}[Entropy is finite]{[B]FINENT}
	All ensembles have a well-defined finite entropy.
\end{condition}

This requirement also applies to the marginal ensembles describing individual DOFs. In the physical theorems below, we use \ref{CST} constrained by \ref{[B]FINENT}: the representation is retained, while ensembles that violate the requirement are excluded.

Connecting statistical mechanics to thermodynamics, the third law motivates a non-negative entropy for physical ensembles. We impose this lower bound on the marginal of each independent DOF, independently of its representation.

\begin{condition}[Lower entropic bound]{[B]ZENT}
	Given a statistical ensemble, the entropy of the marginal of each independent degree of freedom is well-defined and non-negative.
\end{condition}

Under \ref{CST}, this means $S[\rho]=S[\rho_{qp}]\geq0$ for the marginal over an independent DOF. In other coordinates $f,g$, the same invariant entropy is $S[\rho]=S[\rho_{fg}]-E[\ln J_T\mid\rho_{qp}]$. Even finite-entropy ensembles can violate the lower bound: densities concentrated in sufficiently small phase-space regions have negative entropy.

We now state the general uncertainty relationship we want to study.
\begin{condition}[Generalized classical uncertainty relation]{GCUR}
	For each independent DOF and every admissible change of coordinates $(q,p)\mapsto(f,g)$, the uncertainties satisfy $$\sigma_f\sigma_g\geq\frac{\hbar}{e}e^{E[\ln|\{f,g\}|\mid\rho_{qp}]}.$$ 
	Equality is obtained if and only if $S[\rho]=0$ and $\rho_{fg}$ is the product of two Gaussians.
\end{condition}
The product of the physical units of $f,g$ must be the same as for $q,p$. Recall that the coordinate change may map $\mathbb{R}^2$ onto any open coordinate region, including a bounded one, and admissibility requires $E[\left|\ln|\{f,g\}|\right|\mid\rho_{qp}]<\infty$.

Since $\{q,p\}=1$, the relationship for canonical variables reduces to the following.
\begin{condition}[Classical uncertainty relation]{CUR}
	The uncertainty on each independent DOF, in the canonical coordinates $q,p$ of the representation, is bounded by $\sigma_q\sigma_p\geq\frac{\hbar}{e}$. Equality is obtained if and only if position and momentum are independent Gaussians with zero joint entropy.
\end{condition}

\section{Theorems}

We first show that the classical representation alone guarantees neither the physical lower bound on entropy nor the uncertainty relation. We then show that constraining the representation by the entropic lower bound guarantees the uncertainty relation.

\begin{thrm}
	Over \ref{CPS}, neither \ref{[B]ZENT} nor \ref{CUR} is guaranteed.
\end{thrm}

\begin{proof}
	Consider one independent DOF, represented by $\mathbb{R}^2$ with its canonical symplectic form. Let $\rho(q,p)=\frac{1}{2\pi\sigma_q\sigma_p}e^{-\frac{q^2}{2\sigma_q^2}}e^{-\frac{p^2}{2\sigma_p^2}}$, where $\sigma_q=s>0$ and $\sigma_p=\frac{h}{4\pi e s}$. This is a normalized product of two centered Gaussian densities, with standard deviations $\sigma_q$ and $\sigma_p$. It is therefore an ensemble admitted by \ref{CPS}.

	Using the Gibbs/Shannon entropy specified by \ref{CPS}, we find
	\begin{equation*}
		\begin{aligned}
			S[\rho]
			&=\ln\left(\frac{2\pi e\,\sigma_q\sigma_p}{h}\right)=\ln\left(\frac{2\pi e}{h}\frac{h}{4\pi e}\right)
			=-\ln2<0.
		\end{aligned}
	\end{equation*}
	Thus \ref{[B]ZENT} fails. Moreover,
	\begin{equation*}
		\sigma_q\sigma_p
		=\frac{h}{4\pi e}
		=\frac{\hbar}{2e}
		<\frac{\hbar}{e},
	\end{equation*}
	so the canonical uncertainty bound in \ref{CUR} also fails. Since \ref{CPS} admits this ensemble, neither condition is guaranteed.
\end{proof}

We now impose the physical lower bound on entropy.

\begin{thrm}
	Over \ref{CPS} constrained by \ref{[B]ZENT}, condition \ref{CUR} is guaranteed.
\end{thrm}

\begin{proof}
	Let $\rho(q,p)$ be the marginal describing an independent DOF. By \ref{CPS}, its physical entropy is represented by the Gibbs/Shannon formula, and by \ref{[B]ZENT}, $S[\rho]\geq0$. Proposition \ref{entropyVariance} therefore gives
	\begin{equation}\label{rp-cm-classicalUncertaintyRelation}
		\sigma_q\sigma_p
		\geq\frac{h}{2\pi e}e^{S[\rho]}
		\geq\frac{h}{2\pi e}
		=\frac{\hbar}{e}.
	\end{equation}
	Equality holds if and only if the first inequality is saturated by independent Gaussian variables and the second by $S[\rho]=0$. This proves the canonical uncertainty relation and its equality condition. The calculation concerns finite variances; if a variance is infinite, the product is unbounded and the inequality is automatic.

	Now let $f,g$ be variables satisfying the transformation conditions in \ref{CUR}. Proposition \ref{entropyCoordinates} gives $S[\rho_{fg}]=S[\rho_{qp}]+E[\ln|\{f,g\}|]\geq E[\ln|\{f,g\}|]$. Applying Proposition \ref{entropyVariance} to the transformed density gives
	\begin{equation*}
		\begin{aligned}
			\sigma_f\sigma_g
			&\geq\frac{h}{2\pi e}e^{S[\rho_{fg}]} \geq\frac{h}{2\pi e}e^{E[\ln|\{f,g\}|]} =\frac{\hbar}{e}e^{E[\ln|\{f,g\}|]}.
		\end{aligned}
	\end{equation*}
	Again, an infinite variance makes the product unbounded. Thus both uncertainty relations in \ref{CUR} follow.
\end{proof}

\section{Conclusion}

We have found a classical analogue of the quantum uncertainty relation due to the lower bound on entropy. This lower bound is already present in quantum mechanics, where pure states have zero entropy and mixed states have non-negative entropy. The unrestricted classical representation guarantees neither the entropic lower bound nor the uncertainty relation. We conclude that the classical uncertainty relation is a consequence of the entropic lower bound imposed on the classical representation.

\begin{lemma}[Fischer's inequality]\label{FischerInequality}
	Let $M$, $A$, $C$ be positive-semidefinite matrixes such that
	$$M = \begin{bmatrix}
		A & B \\ B^{\dagger} & C
	\end{bmatrix}.$$
	Then $\det(A) \det(B) \geq \det(M)$ with the equality holding if and only if $B = 0$.
\end{lemma}

\begin{defn}
	Let $\rho(\xi^a)$ be a probability distribution over classical phase space with $n$ DOFs. The covariance matrix $\Sigma$ is defined as 	$$\Sigma = [cov(\xi^a, \xi^b)] = \begin{bmatrix}
		\Sigma_{11} & \Sigma_{12} & \dots & \Sigma_{1n} \\
		\Sigma_{21} & \Sigma_{22} & \dots & \Sigma_{2n} \\
		\vdots & \vdots & \ddots & \vdots \\
		\Sigma_{n1} & \Sigma_{n2} & \dots & \Sigma_{nn} \\
	\end{bmatrix},$$
	where $$\Sigma_{ii} = \begin{bmatrix} \sigma^2_{q^i} & cov(q^i,p_i) \\ cov(p_i, q^i) & \sigma^2_{p_i} \end{bmatrix}$$ represents the covariance matrix of the marginals within the $i$-th DOF and $\Sigma_{ij}$ represents the covariance, and therefore the correlation, between two DOFs.
\end{defn}

\begin{prop}
	Given a covariance matrix $\Sigma$, 
	\begin{equation}
		\prod_{i=1}^{n} \sigma^2_{q^i}\sigma^2_{p^i} \geq \prod_{i=1}^{n} \det(\Sigma_{ii}) \geq \Sigma.
	\end{equation}
\end{prop}
\begin{proof}
	The claim follows by applying \ref{FischerInequality} multiple times. Let $\Sigma_{>ii}$ be the block matrix formed by all block matrices with indexes greater than $i$. Then
	\begin{align}
	 \det(M) &\leq \det(\Sigma_{11}) \det(\Sigma_{>22}) \leq \det(\Sigma_{11}) \det(\Sigma_{22}) \det(\Sigma_{>33}) \leq \dots \leq \prod_{i=1}^{n} \det(\Sigma_{ii}).
	\end{align}
	Within each $\Sigma_{ii}$, $\det(\Sigma_{ii}) \leq \det(\sigma^2_{q^i}) \det(\sigma^2_{p_i}) = \sigma^2_{q^i}\sigma^2_{p^i}$.
\end{proof}
