\begin{abstract}
	We show that a zero lower bound on physical entropy, motivated by the third law of thermodynamics, gives rise to the classical uncertainty relation $\sigma_q\sigma_p\geq\frac{\hbar}{e}$. For non-conjugate variables, the relation becomes $\sigma_f\sigma_g\geq\frac{\hbar}{e}e^{E[\ln|\{f,g\}|]}$, where $E[\ln|\{f,g\}|]$ is the expectation of the logarithm of the absolute value of the Poisson bracket.
\end{abstract}

\section{Introduction}

The third law of thermodynamics provides a physical reference for zero entropy. Here we show that imposing a non-negative entropy in classical statistical mechanics gives a classical version of the quantum uncertainty relation:
\begin{equation}
	\sigma_q\sigma_p\geq\frac{h}{2\pi e}=\frac{\hbar}{e}.
\end{equation}
In this context, $h$ is the phase-space area for which a uniform distribution has zero entropy, and $\hbar=\frac{h}{2\pi}$. The $2\pi e$ factor appears because the product of two independent Gaussian distributions minimizes the uncertainty at fixed entropy.

For convenience, entropy is measured in nats, meaning that we use the natural logarithm and set the Boltzmann constant to $k_B=1$. Since we are looking at zero entropy, a change of entropy units does not change the result.

\section{Mathematical prerequisites}

We first establish the mathematical results relating entropy, uncertainty and changes of variables. For a probability density $\rho$ on $\mathbb{R}^{2n}$, define $S[\rho]=-\int\rho\ln(h^n\rho)\,d\mu$, where $h>0$ is fixed, $\mu$ is Lebesgue measure and $0\ln0=0$. The entropy is understood wherever the integral is well-defined, including extended values. We write $E$ for expectation with respect to the density under consideration.

\begin{prop}\label{uniformEntropy}
	Let $\rho_U$ be the uniform probability density over a measurable region $U$ with $0<\mu(U)<\infty$. Then $S[\rho_U]=0$ if and only if $\mu(U)=h^n$.
\end{prop}

\begin{proof}
	The entropy for a uniform distribution reduces to $S[\rho_U]=\ln\frac{\mu(U)}{h^n}$. The logarithm is zero if and only if $\frac{\mu(U)}{h^n}=1$.
\end{proof}

\begin{prop}\label{entropyVariance}
	Let $\sigma(q,p)$ be a probability density with finite variances $\sigma_q^2,\sigma_p^2>0$. Then $S[\sigma]\leq\ln\left(\frac{2\pi e\,\sigma_q\sigma_p}{h}\right)$, or equivalently $\sigma_q\sigma_p\geq\frac{h}{2\pi e}e^{S[\sigma]}$. Equality holds if and only if $q$ and $p$ are independent Gaussian variables. The entropy may take the value $-\infty$, with $e^{-\infty}=0$.
\end{prop}

\begin{proof}
	We first find the stationary densities for the product of the variances at a fixed finite entropy $S$. We use Lagrange multipliers to vary $\sigma_q^2\sigma_p^2=\int(q-\mu_q)^2\rho\,dqdp\int(p-\mu_p)^2\rho\,dqdp$, where $\mu_q$ and $\mu_p$ are the means of $q$ and $p$, under two constraints: $\rho$ integrates to $1$ and its entropy is $S$. We have
	\begin{equation*}
		\begin{aligned}
			L={}&\int(q-\mu_q)^2\rho\,dqdp
			\int(p-\mu_p)^2\rho\,dqdp\\
			&+\lambda_1\left(\int\rho\,dqdp-1\right)
			+\lambda_2\left(-\int\rho\ln(h\rho)\,dqdp-S\right),\\
			\delta L={}&\int\delta\rho\big[
			(q-\mu_q)^2\sigma_p^2+\sigma_q^2(p-\mu_p)^2\\
			&\hspace{5em}+\lambda_1-\lambda_2\ln(h\rho)-\lambda_2
			\big]\,dqdp=0,\\
			\lambda_2\ln(h\rho)={}&\lambda_1-\lambda_2
			+(q-\mu_q)^2\sigma_p^2+\sigma_q^2(p-\mu_p)^2,\\
			\rho={}&\frac{1}{h}
			e^{\frac{\lambda_1-\lambda_2}{\lambda_2}}
			e^{\frac{(q-\mu_q)^2\sigma_p^2}{\lambda_2}}
			e^{\frac{\sigma_q^2(p-\mu_p)^2}{\lambda_2}}.
		\end{aligned}
	\end{equation*}
	Integrability requires $\lambda_2<0$. We recognize the stationary density as the product of two independent Gaussians. Using the standard expression, we find
	\begin{equation*}
		\begin{aligned}
			\rho
			&=\frac{1}{2\pi\sigma_q\sigma_p}
			e^{-\frac{(q-\mu_q)^2}{2\sigma_q^2}}
			e^{-\frac{(p-\mu_p)^2}{2\sigma_p^2}},\\
			S[\rho]
			&=\ln\left(\frac{2\pi\sigma_q\sigma_p}{h}\right)
			+\frac{\sigma_q^2}{2\sigma_q^2}
			+\frac{\sigma_p^2}{2\sigma_p^2}\\
			&=\ln\left(\frac{2\pi e\,\sigma_q\sigma_p}{h}\right),\\
			\sigma_q\sigma_p
			&=\frac{h}{2\pi e}e^{S[\rho]}.
		\end{aligned}
	\end{equation*}

	To verify that this gives the global bound, let $\sigma$ be an arbitrary density with finite variances, and take $\rho$ to be the Gaussian above with the same means and variances as $\sigma$. Write $E[\,\cdot\mid\sigma]$ for expectation with respect to $\sigma$. Consider the KL divergence $D(\sigma\Vert\rho)=\int\sigma\ln\left(\frac{\sigma}{\rho}\right)\,dqdp$. The inequality $u\ln u-u+1\geq0$, with equality only at $u=1$, gives $D\geq0$ by setting $u=\frac{\sigma}{\rho}$ and integrating against $\rho$. Equality holds if and only if $\sigma=\rho$ almost everywhere. Moreover,
	\begin{equation*}
		\begin{aligned}
			D(\sigma\Vert\rho)
			&=\int\sigma\ln(h\sigma)\,dqdp
			-\int\sigma\ln(h\rho)\,dqdp\\
			&=-S[\sigma]
			+\ln\left(\frac{2\pi\sigma_q\sigma_p}{h}\right)
			+\frac{E[(q-\mu_q)^2\mid\sigma]}{2\sigma_q^2}
			+\frac{E[(p-\mu_p)^2\mid\sigma]}{2\sigma_p^2}\\
			&=\ln\left(\frac{2\pi e\,\sigma_q\sigma_p}{h}\right)-S[\sigma]
			\geq0.
		\end{aligned}
	\end{equation*}
	Since $\ln(h\rho(q,p))$ is a quadratic function, the expectation $E[\ln(h\rho(q,p))\mid\sigma]$ is finite for all $\sigma$ with finite variance. Thus the identity is also valid when $D=+\infty$ and $S[\sigma]=-\infty$; finite variances preclude $S[\sigma]=+\infty$. Rearranging gives the stated bound, with equality precisely for the product of two independent Gaussian densities.
\end{proof}

\begin{prop}\label{entropyCoordinates}
	Let $(q,p)\mapsto(f(q,p),g(q,p))$ be a continuously differentiable change of variables with continuously differentiable inverse. Let $\rho_{qp}$ and $\rho_{fg}$ be the two densities of the same probability distribution. If $S[\rho_{qp}]$ is well-defined and $\ln|\{f,g\}|$ is integrable, then $S[\rho_{fg}]=S[\rho_{qp}]+E[\ln|\{f,g\}|]$.
\end{prop}

\begin{proof}
	The Poisson bracket is the Jacobian determinant of the transformation:
	\begin{equation*}
		\{f,g\}
		=\partial_q f\,\partial_p g-\partial_p f\,\partial_q g
		=\left|\begin{array}{cc}
			\partial_q f&\partial_p f\\
			\partial_q g&\partial_p g
		\end{array}\right|.
	\end{equation*}
	Therefore $\rho_{qp}(q,p)=\rho_{fg}(f,g)|\{f,g\}|$. The absolute value allows either orientation. In integrals over $f,g$, functions of $q,p$ are evaluated using the inverse transformation. We have
	\begin{equation*}
		\begin{aligned}
			S[\rho_{qp}]
			&=-\int\rho_{qp}\ln(h\rho_{qp})\,dqdp\\
			&=-\int\rho_{fg}|\{f,g\}|
			\ln\left(h\rho_{fg}|\{f,g\}|\right)\,dqdp\\
			&=-\int\rho_{fg}
			\ln\left(h\rho_{fg}|\{f,g\}|\right)\,dfdg\\
			&=-\int\rho_{fg}\ln(h\rho_{fg})\,dfdg
			-\int\rho_{fg}\ln|\{f,g\}|\,dfdg\\
			&=S[\rho_{fg}]-E[\ln|\{f,g\}|].
		\end{aligned}
	\end{equation*}
	The logarithmic expectation is finite by assumption, so this identity also holds for extended entropy values. Rearranging proves the result.
\end{proof}

\section{Conditions}

We now state the conditions for the theorems, starting with the classical phase-space representation. This includes pure states, ensembles, configuration counting and entropy.

\begin{condition}[Classical phase space]{CPS}
	The state space of a classical system with $n$ degrees of freedom (DOFs) is represented by $M=\mathbb{R}^{2n}$ with canonical coordinates $(q^i,p_i)$ and symplectic form $\omega=\sum_i dq^i\wedge dp_i$. The count of configurations for each DOF is represented by its canonical phase-space area, and the count of states by the Liouville measure $\mu(U)=\int_U dq^1\cdots dq^n dp_1\cdots dp_n=\int_U\frac{\omega^n}{n!}$.

	A statistical ensemble is represented by a probability density $\rho(q^i,p_i)$ relative to the Liouville measure, and statistical mixtures by convex combinations of densities. The entropy is given by the Gibbs/Shannon formula $S[\rho]=-\int_M\rho\ln(h^n\rho)\,d\mu$, where $h>0$ is a fixed constant with units of phase-space area.
\end{condition}

The constant $h$ makes the entropy formula dimensionally correct and fixes its zero. By Proposition \ref{uniformEntropy}, a uniform ensemble has zero entropy exactly when its phase-space volume is $h^n$.

Connecting statistical mechanics to thermodynamics, the third law of thermodynamics will impose that the entropy of all statistical ensembles must be non-negative. This also includes the entropy of every marginal of each independent degree of freedom. We state this requirement independently of its representation.

\begin{condition}[Lower entropic bound]{[B]ZENT}
	Given a statistical ensemble, the entropy of the marginal of each independent degree of freedom is well-defined and non-negative.
\end{condition}

Under \ref{CPS}, this means $S[\rho(q,p)]\geq0$ for the marginal describing the independent DOF. The representation does not guarantee this requirement: densities concentrated in sufficiently small phase-space regions have negative entropy. We therefore constrain the admissible ensembles by \ref{[B]ZENT}.

The goal is to show that this constraint guarantees the following uncertainty relation. The formulas apply whenever the indicated variances and expectations are well-defined.

\begin{condition}[Classical uncertainty relation]{CUR}
	The uncertainty on each independent DOF is bounded by $\sigma_q\sigma_p\geq\frac{\hbar}{e}$, where $\hbar=\frac{h}{2\pi}$. Equality is obtained if and only if position and momentum are independent Gaussian variables with zero joint entropy.

	If $f$ and $g$ are two variables that span the same DOF through a continuously differentiable invertible transformation with continuously differentiable inverse, and the product of their physical units is the same as for $q,p$, then $\sigma_f\sigma_g\geq\frac{\hbar}{e}e^{E[\ln|\{f,g\}|]}$, provided the logarithm of the absolute Poisson bracket has finite expectation.
\end{condition}

\section{Theorems}

We first show that the classical representation alone guarantees neither the physical lower bound on entropy nor the uncertainty relation. We then show that constraining the representation by the entropic lower bound guarantees the uncertainty relation.

\begin{thrm}
	Over \ref{CPS}, neither \ref{[B]ZENT} nor \ref{CUR} is guaranteed.
\end{thrm}

\begin{proof}
	Consider one independent DOF, represented by $\mathbb{R}^2$ with its canonical symplectic form. Let $\rho(q,p)=\frac{1}{2\pi\sigma_q\sigma_p}e^{-\frac{q^2}{2\sigma_q^2}}e^{-\frac{p^2}{2\sigma_p^2}}$, where $\sigma_q=s>0$ and $\sigma_p=\frac{h}{4\pi e s}$. This is a normalized product of two centered Gaussian densities, with standard deviations $\sigma_q$ and $\sigma_p$. It is therefore an ensemble admitted by \ref{CPS}.

	Using the Gibbs/Shannon entropy specified by \ref{CPS}, we find
	\begin{equation*}
		\begin{aligned}
			S[\rho]
			&=\ln\left(\frac{2\pi e\,\sigma_q\sigma_p}{h}\right)=\ln\left(\frac{2\pi e}{h}\frac{h}{4\pi e}\right)
			=-\ln2<0.
		\end{aligned}
	\end{equation*}
	Thus \ref{[B]ZENT} fails. Moreover,
	\begin{equation*}
		\sigma_q\sigma_p
		=\frac{h}{4\pi e}
		=\frac{\hbar}{2e}
		<\frac{\hbar}{e},
	\end{equation*}
	so the canonical uncertainty bound in \ref{CUR} also fails. Since \ref{CPS} admits this ensemble, neither condition is guaranteed.
\end{proof}

We now impose the physical lower bound on entropy.

\begin{thrm}
	Over \ref{CPS} constrained by \ref{[B]ZENT}, condition \ref{CUR} is guaranteed.
\end{thrm}

\begin{proof}
	Let $\rho(q,p)$ be the marginal describing an independent DOF. By \ref{CPS}, its physical entropy is represented by the Gibbs/Shannon formula, and by \ref{[B]ZENT}, $S[\rho]\geq0$. Proposition \ref{entropyVariance} therefore gives
	\begin{equation}\label{rp-cm-classicalUncertaintyRelation}
		\sigma_q\sigma_p
		\geq\frac{h}{2\pi e}e^{S[\rho]}
		\geq\frac{h}{2\pi e}
		=\frac{\hbar}{e}.
	\end{equation}
	Equality holds if and only if the first inequality is saturated by independent Gaussian variables and the second by $S[\rho]=0$. This proves the canonical uncertainty relation and its equality condition. The calculation concerns finite variances; if a variance is infinite, the product is unbounded and the inequality is automatic.

	Now let $f,g$ be variables satisfying the transformation conditions in \ref{CUR}. Proposition \ref{entropyCoordinates} gives $S[\rho_{fg}]=S[\rho_{qp}]+E[\ln|\{f,g\}|]\geq E[\ln|\{f,g\}|]$. Applying Proposition \ref{entropyVariance} to the transformed density gives
	\begin{equation*}
		\begin{aligned}
			\sigma_f\sigma_g
			&\geq\frac{h}{2\pi e}e^{S[\rho_{fg}]} \geq\frac{h}{2\pi e}e^{E[\ln|\{f,g\}|]} =\frac{\hbar}{e}e^{E[\ln|\{f,g\}|]}.
		\end{aligned}
	\end{equation*}
	Again, an infinite variance makes the product unbounded. Thus both uncertainty relations in \ref{CUR} follow.
\end{proof}

\section{Conclusion}

We have found a classical analogue of the quantum uncertainty relation due to the lower bound on entropy. This lower bound is already present in quantum mechanics, where pure states have zero entropy and mixed states have non-negative entropy. The unrestricted classical representation guarantees neither the entropic lower bound nor the uncertainty relation. We conclude that the classical uncertainty relation is a consequence of the entropic lower bound imposed on the classical representation.

\begin{lemma}[Fischer's inequality]\label{FischerInequality}
	Let $M$, $A$, $C$ be positive-semidefinite matrixes such that
	$$M = \begin{bmatrix}
		A & B \\ B^{\dagger} & C
	\end{bmatrix}.$$
	Then $\det(A) \det(B) \geq \det(M)$ with the equality holding if and only if $B = 0$.
\end{lemma}

\begin{defn}
	Let $\rho(\xi^a)$ be a probability distribution over classical phase space with $n$ DOFs. The covariance matrix $\Sigma$ is defined as 	$$\Sigma = [cov(\xi^a, \xi^b)] = \begin{bmatrix}
		\Sigma_{11} & \Sigma_{12} & \dots & \Sigma_{1n} \\
		\Sigma_{21} & \Sigma_{22} & \dots & \Sigma_{2n} \\
		\vdots & \vdots & \ddots & \vdots \\
		\Sigma_{n1} & \Sigma_{n2} & \dots & \Sigma_{nn} \\
	\end{bmatrix},$$
	where $$\Sigma_{ii} = \begin{bmatrix} \sigma^2_{q^i} & cov(q^i,p_i) \\ cov(p_i, q^i) & \sigma^2_{p_i} \end{bmatrix}$$ represents the covariance matrix of the marginals within the $i$-th DOF and $\Sigma_{ij}$ represents the covariance, and therefore the correlation, between two DOFs.
\end{defn}

\begin{prop}
	Given a covariance matrix $\Sigma$, 
	\begin{equation}
		\prod_{i=1}^{n} \sigma^2_{q^i}\sigma^2_{p^i} \geq \prod_{i=1}^{n} \det(\Sigma_{ii}) \geq \Sigma.
	\end{equation}
\end{prop}
\begin{proof}
	The claim follows by applying \ref{FischerInequality} multiple times. Let $\Sigma_{>ii}$ be the block matrix formed by all block matrices with indexes greater than $i$. Then
	\begin{align}
	 \det(M) &\leq \det(\Sigma_{11}) \det(\Sigma_{>22}) \leq \det(\Sigma_{11}) \det(\Sigma_{22}) \det(\Sigma_{>33}) \leq \dots \leq \prod_{i=1}^{n} \det(\Sigma_{ii}).
	\end{align}
	Within each $\Sigma_{ii}$, $\det(\Sigma_{ii}) \leq \det(\sigma^2_{q^i}) \det(\sigma^2_{p_i}) = \sigma^2_{q^i}\sigma^2_{p^i}$.
\end{proof}
