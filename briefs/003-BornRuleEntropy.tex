\begin{abstract}
	We show that the following conditions are equivalent: conditional probability is given by the Born rule, the entropy is given by the von Neumann entropy and mutual exclusivity corresponds to orthogonality.
\end{abstract}

\section{Introduction}

In quantum mechanics, the Born rule
\begin{equation}
	p(\psi|\phi) = \frac{\< \phi | \psi \>\< \psi | \phi \>}{\< \psi | \psi \>\< \phi | \phi \>}
\end{equation}
defines conditional probability for two states: the probability of measuring state $\psi$ given that $\phi$ was prepared. Similarly, the von Neumann entropy
\begin{equation}
	S(\rho) = - \tr(\rho \log \rho)
\end{equation}
defines the entropy for quantum ensembles. Here we show that once one is assumed, the other is implied. Moreover, we also show that just requiring that mutual exclusivity corresponds to orthogonality is also equivalent to those definitions, and therefore represents a conceptually simpler and crisper requirement.

\section{Mathematical prerequisite}

We prove a useful mathematical result that follows from the affine structure of euclidean balls.

\begin{prop}\label{affineBall}
	Let $B=\{r\in\mathbb{R}^n:\|r\|\leq1\}$ be an n-dimensional ball and let $u\in\mathbb{R}^n$ with $\|u\|=1$ be a point on the surface. If $f:B\to[0,1]$ is an affine function of $B$, with $f(u)=1$ and $f(-u)=0$, then
	\begin{equation*}
		f(r)=\frac{1+u\cdot r}{2}.
	\end{equation*}
\end{prop}

\begin{proof}
	By affinity, write $f(r)=c+d\cdot r$ for some $c \in \mathbb{R}$ and $d \in \mathbb{R}^n$. Since $f(u)\neq f(-u)$, we have $d\neq0$. By Cauchy--Schwarz, the minimum and maximum of $d\cdot r$ on $B$ are $-\|d\|$ and $\|d\|$, attained at $r=-\frac{d}{\|d\|}$ and $r=\frac{d}{\|d\|}$ respectively. Since $f$ takes values in $[0,1]$ and attains both endpoints, we have $c-\|d\|=0$ and $c+\|d\|=1$. Thus $c=\|d\|=\frac{1}{2}$. Moreover, $f(u)=1$ gives $d\cdot u=\frac{1}{2}$, so
	\begin{equation*}
		\|2d-u\|^2
		=4\|d\|^2-4d\cdot u+\|u\|^2
		=1-2+1=0.
	\end{equation*}
	Therefore $d=\frac{u}{2}$, and
	\begin{equation*}
		f(r)=c+d\cdot r=\frac{1+u\cdot r}{2}.
	\end{equation*}
\end{proof}

\section{Conditions}

We now state the conditions for the theorem, starting with the background conditions. The following condition provides the standard Hilbert space representation without specifying the Born rule or the entropy.

\begin{condition}[Ensembles and quantum states]{QE-DENS}
	The state space of a quantum system is represented by the projective space $\mathcal{P}(\mathcal{H})$ of a separable complex Hilbert space $\mathcal{H}$. The ensemble space is represented by the density operators $\mathcal{D}(\mathcal{H})$.
\end{condition}
In terms of notation, we will use $\psi \in \mathcal{P}(\mathcal{H})$ for states, $|\psi\> \in \mathcal{H}$ for (typically normalized) vector representatives and $\rho_{\psi} \in \mathcal{D}(\mathcal{H})$ for a density operator representing a pure state.

The following condition states a standard rule for conditional probabilities.

\begin{condition}[Probabilities of statistical mixtures]{[B]PMIX}
	The conditional probability of a statistical mixture is the statistical mixture of the conditional probabilities. That is,
	$p(\psi\mid\sum_i \lambda_i\rho_i)=\sum_i \lambda_i p(\psi\mid\rho_i)$.
\end{condition}

The following condition makes the properties of mutual exclusivity explicit. The concept itself is the idea that only one outcome can happen at a time. This can be characterized in terms of probabilities and entropy.

\begin{condition}[Mutual exclusivity conditions]{[B]MUTEX}
	Let $\psi_i \in \mathcal{P}$ be a countable sequence of states. Then the following are equivalent:
	\begin{itemize}
		\item $\psi_i$ are mutually exclusive
		\item $p(\psi_i | \psi_j) = \delta_{ij}$
		\item the entropy obeys $S(\sum_i p_i \rho_{\psi_i}) = - \sum_i p_i \log p_i$ for all $p_i$ such that $\sum_i p_i = 1$.
	\end{itemize}
\end{condition}

We now state the conditions we will want to prove equivalent.

\begin{condition}[Conditional probability from Born rule]{BR-BORN}
	The conditional probability between two states is given by the Born rule. That is, $p(\psi|\phi) = \frac{\< \phi | \psi \>\< \psi | \phi \>}{\< \psi | \psi \>\< \phi | \phi \>}$.
\end{condition}

\begin{condition}[Von Neumann entropy]{BR-ENT}
	The entropy function is given by the von Neumann entropy. That is, $S(\rho) = - \tr(\rho \log \rho)$.
\end{condition}

\begin{condition}[Orthogonality is mutual exclusivity]{BR-ORME}
	Mutual exclusivity is represented by orthogonality. That is, $\psi$ and $\phi$ are mutually exclusive if $\<\psi | \phi \> = 0$.
\end{condition}

\section{Theorem}

We are now ready to state and prove the theorem.

\begin{thrm}
	Over \ref{QE-DENS}+\ref{[B]MUTEX}+\ref{[B]PMIX}, conditions \ref{BR-BORN}, \ref{BR-ENT} and \ref{BR-ORME} are equivalent and are not guaranteed.
\end{thrm}

\begin{proof}
	\ref{BR-BORN} $\implies$ \ref{BR-ORME}.
	Let $\psi,\phi$ be orthogonal pure states. By \ref{BR-BORN}, $p(\psi|\psi)=p(\phi|\phi)=1$, and $p(\psi|\phi)=p(\phi|\psi)=0$.
	By \ref{[B]MUTEX}, the two preparations are mutually exclusive. Therefore orthogonal pure states are mutually exclusive. This means that \ref{BR-BORN} implies \ref{BR-ORME}.
	
	\ref{BR-ORME} $\implies$ \ref{BR-BORN}.
	Consider two pure states $\psi,\phi$ and a two-dimensional subspace containing representatives of both. Let $\psi^\perp$ be the state orthogonal to $\psi$ in this subspace. By \ref{BR-ORME}, $\psi$ and $\psi^\perp$ are mutually exclusive. Condition \ref{[B]MUTEX} therefore gives
	\begin{equation*}
		p(\psi|\psi)=1,
		\qquad
		p(\psi|\psi^\perp)=0.
	\end{equation*}
	By \ref{[B]PMIX}, $p(\psi|\rho)$ is an affine function on the corresponding Bloch ball, taking values in $[0,1]$. It takes the values $1$ and $0$ at the opposite poles $\psi$ and $\psi^\perp$. Proposition \ref{affineBall} gives
	\begin{equation*}
		\begin{aligned}
			p(\psi|\phi)
			&=\frac{1+\cos\theta_{\psi\phi}}{2}
			=\cos^2\left(\frac{\theta_{\psi\phi}}{2}\right)
			=\frac{\<\phi|\psi\>\<\psi|\phi\>}
			{\<\psi|\psi\>\<\phi|\phi\>},
		\end{aligned}
	\end{equation*}
	where $\theta_{\psi\phi}$ is the angle between their Bloch vectors. This includes coincident states, for which $\theta_{\psi\phi}=0$. Therefore the conditional probability between two states is given by the Born rule. Which means \ref{BR-ORME} implies \ref{BR-BORN}.
	
	\ref{BR-ORME} $\implies$ \ref{BR-ENT}.
	Let $\rho$ be an arbitrary density operator. Choose an orthonormal basis of eigenvectors and write its spectral decomposition as
	\begin{equation*}
		\rho=\sum_i\lambda_i\rho_{\psi_i},
		\qquad
		\lambda_i\geq0,
		\qquad
		\sum_i\lambda_i=1.
	\end{equation*}
	The states $\psi_i$ are pairwise orthogonal. By \ref{BR-ORME} and \ref{[B]MUTEX}, $p(\psi_i|\psi_j)=\delta_{ij}$. Applying \ref{[B]MUTEX} to the family then gives
	\begin{equation*}
		\begin{aligned}
			S(\rho)
			&=S\left(\sum_i\lambda_i\rho_{\psi_i}\right)
			=-\sum_i\lambda_i\log\lambda_i =-\tr(\rho\log\rho).
		\end{aligned}
	\end{equation*}
	Here $0\log0=0$, and the equality also holds when the entropy is infinite. Since $\rho$ was arbitrary, the entropy is given by the von Neumann entropy. Which means \ref{BR-ORME} implies \ref{BR-ENT}.
	
	\ref{BR-ENT} $\implies$ \ref{BR-ORME}.
	Let $\psi,\phi$ be orthogonal pure states. For every $\lambda\in[0,1]$, the mixture $\rho_\lambda=\lambda\rho_\psi+(1-\lambda)\rho_\phi$ has eigenvalues $\lambda$ and $1-\lambda$, with any remaining eigenvalues zero. By \ref{BR-ENT}, $S(\rho_\lambda) = - \lambda \log\lambda - (1-\lambda) \log(1-\lambda)$. Thus the entropy of their mixtures obeys the Shannon formula for every mixing weight. By \ref{[B]MUTEX}, $\psi$ and $\phi$ are mutually exclusive. Therefore \ref{BR-ENT} implies \ref{BR-ORME}.
	
	To show that the conditions are not guaranteed, it suffices to show a mathematical model that satisfies the background conditions while not satisfying the additional conditions. We do not need to show that this mathematical model represents a valid physical system. The general idea is to construct a model where all states have some intrinsic noise, so that no mutually exclusive pairs of states exist and perfect distinguishability fails. Consider
	$\mathcal{H}=\mathbb{C}^2$, with pure states represented by the
	Bloch sphere and ensembles by the full space
	$\mathcal{D}(\mathcal{H})$. Thus \ref{QE-DENS} holds.
	
	Define
	\begin{equation*}
		f(\rho)=\frac{I}{4}+\frac{\rho}{2},
		\qquad
		h(t)=-t\log t-(1-t)\log(1-t),
		\qquad
		h_0=h\left(\frac{1}{4}\right).
	\end{equation*}
	Assign conditional probabilities and entropy by
	\begin{equation*}
		\begin{aligned}
			p(\psi|\rho)
			&=\frac{4}{3}\tr\left(\rho_\psi f(\rho)\right)
			=\frac{1}{3}+\frac{2}{3}\tr(\rho_\psi\rho),\\
			S(\rho)
			&=-\tr\left(f(\rho)\log f(\rho)\right)-h_0.
		\end{aligned}
	\end{equation*}
	The subtraction removes the constant entropy introduced by
	the noise on pure preparations.
	
	These are valid binary-test probabilities. For each $\psi$,
	define the available test by the two effects
	\begin{equation*}
		E_\psi=\frac{I}{3}+\frac{2}{3}\rho_\psi,
		\qquad
		I-E_\psi=\frac{2}{3}(I-\rho_\psi).
	\end{equation*}
	The eigenvalues of $E_\psi$ are $1$ and $\frac{1}{3}$, so
	$0\leq E_\psi\leq I$. The two outcome probabilities are
	$\tr(E_\psi\rho)=p(\psi|\rho)$ and $1-p(\psi|\rho)$.
	Take these tests and their classical randomizations and
	post-processings as the available measurements. None can
	perfectly distinguish two preparations: the first outcome
	always has probability at least $\frac{1}{3}$, while the
	second never has probability greater than $\frac{2}{3}$.
	Randomization and post-processing cannot give perfect
	discrimination when none of the constituent tests does.
	Thus there are no mutually exclusive pairs of preparations.
	
	We first verify \ref{[B]PMIX}. For any finite or countable
	statistical mixture, with $q_i\geq0$ and $\sum_iq_i=1$,
	\begin{equation*}
		\begin{aligned}
			p\left(\psi\middle|\sum_iq_i\rho_i\right)
			=\frac{1}{3}
			+\frac{2}{3}\sum_iq_i\tr(\rho_\psi\rho_i)
			=\sum_iq_i\left(
			\frac{1}{3}+\frac{2}{3}\tr(\rho_\psi\rho_i)
			\right)
			=\sum_iq_i p(\psi|\rho_i).
		\end{aligned}
	\end{equation*}
	Therefore \ref{[B]PMIX} holds. Moreover,
	\begin{equation*}
		\frac{1}{3}\leq p(\psi|\rho)\leq1,
		\qquad p(\psi|\psi)=1.
	\end{equation*}
	
	Next, we establish the entropy bounds. If $\rho$ has
	eigenvalues $t$ and $1-t$, then $f(\rho)$ has eigenvalues
	$\frac{1}{4}+\frac{t}{2}$ and
	$\frac{3}{4}-\frac{t}{2}$. Both lie in
	$[\frac{1}{4},\frac{3}{4}]$. Since $h$ is symmetric about
	$\frac{1}{2}$ and increases on $[0,\frac{1}{2}]$,
	\begin{equation*}
		h_0\leq-\tr\left(f(\rho)\log f(\rho)\right)\leq\log2.
	\end{equation*}
	Consequently,
	\begin{equation*}
		0\leq S(\rho)\leq\log2-h_0<\log2.
	\end{equation*}
	For every pure preparation, the eigenvalues of $f(\rho_\psi)$
	are $\frac{3}{4}$ and $\frac{1}{4}$, so
	\begin{equation*}
		S(\rho_\psi)=h_0-h_0=0.
	\end{equation*}
	
	We can now verify \ref{[B]MUTEX} for every nonempty finite
	or countable family. For a singleton family, mutual
	exclusivity is trivial, the probability table is
	$p(\psi|\psi)=1$, and its entropy is
	$S(\rho_\psi)=0=-1\log1$. All three criteria hold.
	
	For a family with at least two members, no pair is mutually
	exclusive. The probability-table criterion also fails,
	because
	\begin{equation*}
		p(\psi_i|\psi_j)\geq\frac{1}{3}>0
		\qquad\text{for }i\neq j.
	\end{equation*}
	Finally, choose two indices and assign them weights
	$\frac{1}{2}$, with all remaining weights zero. Then
	\begin{equation*}
		S\left(\frac{1}{2}\rho_{\psi_i}
		+\frac{1}{2}\rho_{\psi_j}\right)
		\leq\log2-h_0
		<\log2
		=-2\frac{1}{2}\log\frac{1}{2}.
	\end{equation*}
	Thus the entropy does not obey the Shannon formula for
	every choice of weights. All three criteria fail for
	every such family. Therefore \ref{[B]MUTEX} holds.
	
	Nevertheless, all three additional conditions fail.
	For orthogonal $\psi,\phi$,
	\begin{equation*}
		p(\psi|\phi)=\frac{1}{3}\neq0,
	\end{equation*}
	so \ref{BR-BORN} fails. Those orthogonal states are not
	mutually exclusive, so \ref{BR-ORME} fails. Finally,
	\begin{equation*}
		S\left(\frac{I}{2}\right)
		=\log2-h_0
		\neq\log2
		=-\tr\left(\frac{I}{2}\log\frac{I}{2}\right),
	\end{equation*}
	so \ref{BR-ENT} fails.
	
	This model satisfies
	\ref{QE-DENS}+\ref{[B]PMIX}+\ref{[B]MUTEX}
	while violating \ref{BR-BORN}, \ref{BR-ORME} and
	\ref{BR-ENT}. Therefore these conditions are not guaranteed.
\end{proof}

\section{Conclusion}

Once we identify the space of ensembles with the ones given by the density operators, assuming the Born rule, the von Neumann entropy or mutual exclusivity for orthogonal states is equivalent. The main extra requirement is the consistency between mutual exclusivity as defined over the conditional probability (i.e. $\psi$ and $\phi$ are mutually exclusive if and only if preparing one gives zero probability of having prepared the other) or over the entropy (i.e. $\phi$ and $\psi$ are mutually exclusive if and only if the entropy of all their statistical mixtures is the Shannon entropy). These conditions are additional, and are not implied by the ensemble space of density operators.
